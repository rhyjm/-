    let deck = [];
    let currentIndex = 0;
    let score = 0;
    let streak = 0;
    let answeredCount = 0;
    let wrongCount = 0;
    let showHint = true;
    let isTransitioning = false;

    const symbolDisplay = document.getElementById('symbol-display');
    const zhText = document.getElementById('zh-text');
    const cardCounter = document.getElementById('card-counter');
    const progressBar = document.getElementById('progress-bar');
    const answerInput = document.getElementById('answer-input');
    const feedbackMsg = document.getElementById('feedback-msg');
    const correctReveal = document.getElementById('correct-reveal');
    const scoreCounter = document.getElementById('score-counter');
    const streakCounter = document.getElementById('streak-counter');
    const finishModal = document.getElementById('finish-modal');
    const toggleHintBtn = document.getElementById('toggle-hint-btn');

    function shuffle(array) {
      const arr = [...array];
      for (let i = arr.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [arr[i], arr[j]] = [arr[j], arr[i]];
      }
      return arr;
    }

    function initPractice() {
      deck = shuffle(ELEMENTS);
      currentIndex = 0;
      score = 0;
      streak = 0;
      answeredCount = 0;
      wrongCount = 0;
      updateHeaderStats();
      renderCurrentCard();
    }

    function renderCurrentCard() {
      if (currentIndex >= deck.length) {
        showCompletionModal();
        return;
      }

      isTransitioning = false;
      const current = deck[currentIndex];

      symbolDisplay.textContent = current.symbol;
      zhText.textContent = current.zh;
      cardCounter.textContent = `${currentIndex + 1} / ${deck.length}`;

      const progressPercent = (currentIndex / deck.length) * 100;
      progressBar.style.width = `${progressPercent}%`;

      answerInput.value = '';
      answerInput.className = "input-box";
      feedbackMsg.style.opacity = '0';
      correctReveal.style.opacity = '0';
      answerInput.focus();
    }

    function checkAnswer() {
      if (isTransitioning || currentIndex >= deck.length) return;

      const userRaw = answerInput.value.trim().toLowerCase();
      if (!userRaw) return;

      const current = deck[currentIndex];
      const correctAns = current.ans.toLowerCase();
      const validAnswers = [correctAns];

      if (current.aliases) {
        current.aliases.forEach(a => validAnswers.push(a.toLowerCase()));
      }

      const isCorrect = validAnswers.includes(userRaw);

      isTransitioning = true;
      answeredCount++;

      if (isCorrect) {
        score++;
        streak++;
        updateHeaderStats();

        answerInput.className = "input-box correct";
        feedbackMsg.textContent = "✓ 完全正確！";
        feedbackMsg.style.color = "#34d399";
        feedbackMsg.style.opacity = "1";
        correctReveal.textContent = `英文全名：${current.ans}`;
        correctReveal.style.opacity = "1";

        setTimeout(() => {
          currentIndex++;
          renderCurrentCard();
        }, 600);
      } else {
        streak = 0;
        wrongCount++;
        updateHeaderStats();

        answerInput.className = "input-box wrong";
        feedbackMsg.textContent = "✕ 拼寫有誤";
        feedbackMsg.style.color = "#f87171";
        feedbackMsg.style.opacity = "1";
        correctReveal.textContent = `正解為：${current.ans}`;
        correctReveal.style.color = "#fcd34d";
        correctReveal.style.opacity = "1";

        setTimeout(() => {
          currentIndex++;
          renderCurrentCard();
        }, 1400);
      }
    }

    function skipCard() {
      if (isTransitioning || currentIndex >= deck.length) return;
      const current = deck[currentIndex];
      wrongCount++;
      answeredCount++;
      streak = 0;
      updateHeaderStats();

      feedbackMsg.textContent = "已跳過此題";
      feedbackMsg.style.color = "#fbbf24";
      feedbackMsg.style.opacity = "1";
      correctReveal.textContent = `正確全名為：${current.ans}`;
      correctReveal.style.color = "#fde68a";
      correctReveal.style.opacity = "1";

      isTransitioning = true;
      setTimeout(() => {
        currentIndex++;
        renderCurrentCard();
      }, 850);
    }

    function shuffleAndRestart() {
      initPractice();
      feedbackMsg.textContent = "題庫已重新隨機打亂！";
      feedbackMsg.style.color = "#38bdf8";
      feedbackMsg.style.opacity = "1";
      setTimeout(() => {
        if (!isTransitioning) feedbackMsg.style.opacity = "0";
      }, 1000);
    }

    function toggleChineseHint() {
      showHint = !showHint;
      zhText.style.display = showHint ? 'inline' : 'none';
      toggleHintBtn.innerHTML = showHint 
        ? '<span>隱藏中文提示</span> 👁️' 
        : '<span>顯示中文提示</span> 🙈';
      answerInput.focus();
    }

    function updateHeaderStats() {
      scoreCounter.textContent = `得分: ${score} / ${answeredCount}`;
      streakCounter.textContent = `連對: ${streak} 🔥`;
    }

    function showCompletionModal() {
      const total = deck.length;
      const acc = total > 0 ? Math.round((score / total) * 100) : 0;
      document.getElementById('final-total').textContent = total;
      document.getElementById('final-correct').textContent = score;
      document.getElementById('final-wrong').textContent = wrongCount;
      document.getElementById('final-acc').textContent = `${acc}%`;
      finishModal.classList.add('show');
    }

    function restartPractice() {
      finishModal.classList.remove('show');
      initPractice();
    }

    answerInput.addEventListener('keydown', (e) => {
      if (e.key === 'Enter') {
        e.preventDefault();
        checkAnswer();
      }
    });

    document.addEventListener('keydown', (e) => {
      if (e.ctrlKey && e.code === 'Space') {
        e.preventDefault();
        skipCard();
      }
    });

    window.addEventListener('DOMContentLoaded', () => {
      initPractice();
    });
